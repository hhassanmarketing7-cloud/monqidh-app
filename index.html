from flask import Flask, render_template_string, request app = Flask(__name__) HTML_TEMPLATE = """ منصة الأمن السيبراني الآلية

نظام الفحص والاختراق الآلي


{% if results %}

حالة التنفيذ:

الشركة المستهدفة: {{ results.target }}

الحالة: {{ results.status }}

{% endif %}
""" @app.route("/", methods=["GET", "POST"]) def home(): results = None if request.method == "POST": # استقبال اسم الشركة المدخل من الموقع company_name = request.form.get("company") # التأكد من وصول الاسم ومعالجته برمجياً if company_name: print(f"[!] تم استلام الهدف بنجاح: {company_name}") results = { "target": company_name, "status": "تم استقبال الهدف بنجاح وجاري تمريره لوحدات الفحص والتحليل في الخلفية." } return render_template_string(HTML_TEMPLATE, results=results) if __name__ == "__main__": app.run(host="0.0.0.0", port=5000, debug=True)
